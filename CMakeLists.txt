cmake_minimum_required(VERSION 3.20)
project(XrealWindowsStableScreen VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(XrealStableScreen WIN32 src/main.cpp src/graphics.cpp src/glasses.cpp tests/self_test.cpp)
target_compile_definitions(XrealStableScreen PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(XrealStableScreen PRIVATE /utf-8 /W4 /permissive- /EHsc)
target_link_libraries(XrealStableScreen PRIVATE d3d11 dxgi d3dcompiler setupapi hid user32 gdi32 comctl32 ole32 windowscodecs shell32)
target_sources(XrealStableScreen PRIVATE app.manifest)
target_link_options(XrealStableScreen PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
